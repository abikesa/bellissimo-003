function parseCSV(data) {
const rows = data.split('\n').slice(1);
return rows.map(row => {
const [id, name, waitDays, bloodType] = row.split(',');
return { id, name, waitDays: parseInt(waitDays), bloodType };
});
}
function generateHopeMetric(patient) {
return Math.min(10, 5 + patient.waitDays / 60).toFixed(1);
}
